After — one typed table

type FailurePattern = {
  id: string;
  matches: (step, exit) => boolean;
  suggestion: string;
  explanation: string | null;
  fix: { file; before; after } | null;
};

const failurePatterns: FailurePattern[] = [
  { id: "commonjs-esm-mismatch",
    matches: ..., suggestion, explanation, fix },
  { id: "eslint-violations",
    matches: ..., suggestion, explanation, fix },
  { id: "oom-kill",
    matches: (_, exit) => exit === 137,
    ..., explanation: null, fix: null },
];

function matchFailurePattern(step, job) {
  const exit = job?.exitCode ?? step?.exitCode;
  return failurePatterns.find(p => p.matches(step, exit)) ?? null;
}
Why this matters →
  • Before: change a label in one place and two other places break silently
  • After: one entry per failure type. Change a label and nothing breaks.
  • Future: a real AI service plugs into the same table — same shape
